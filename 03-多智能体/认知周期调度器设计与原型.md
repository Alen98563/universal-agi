## 零、问题

### 0.1 为什么需要认知调度器

五层心智模型不是静态管道 — 每层有独立的计算预算、消息队列和时延约束。调度器的角色：

1. **时间分片**: 在 ~50ms 认知周期内编排多层处理
2. **资源分配**: 按优先级分配 GPU/CPU 时间 (公理 A8)
3. **消息路由**: 确保层级间协议消息按时送达 (消息协议 v1.0)
4. **故障隔离**: 一层超时不阻塞全局
5. **紧急降级**: 生存威胁 → 跳过 L3-L5 直接 L1→L2 反射弧

### 0.2 关键约束

| 约束 | 值 | 来源 |
|------|-----|------|
| 认知周期 | T_cycle ≤ 50ms | 白皮书 v0.2 |
| L1 最大延迟 | 5ms (实时传感器) | 硬件 |
| L2 最大延迟 | 10ms (感知-行动环) | 控制带宽 |
| L3 最大延迟 | 25ms (世界模型更新) | 设计文档 P1 |
| L4 最大延迟 | 40ms (因果推理) | 复杂度 |
| L5 最大延迟 | 48ms (元认知) | 最内层 |
| 紧急反射延迟 | < 20ms | L1→L2 直连 |

### 2.1 认知周期主循环

```python
# src/scheduler.py

import asyncio
import time
import torch
import numpy as np
from dataclasses import dataclass, field
from typing import Dict, List, Optional, Callable, Any
from enum import Enum, auto
from collections import deque
import logging

logger = logging.getLogger(__name__)


class CycleMode(Enum):
    NORMAL = auto()       # 正常五层全周期
    URGENT = auto()       # 紧急: L1→L2 短路
    IDLE = auto()         # 空闲: 无新感知, L3 内部模拟
    REFLECTIVE = auto()   # 反思: L5 独占长周期


@dataclass
class CycleMetrics:
    """单个认知周期的计时和资源度量"""
    cycle_id: int
    start_time: float
    mode: CycleMode

    # 每层的执行时间 (ms)
    layer_times: Dict[int, float] = field(default_factory=dict)

    # 消息统计
    messages_sent: int = 0
    messages_dropped: int = 0

    # 资源
    gpu_util_pct: float = 0.0
    surprise: float = 0.0  # L2 报告的 prediction error 幅度

    def total_time_ms(self) -> float:
        return sum(self.layer_times.values())

    def deadline_met(self) -> bool:
        return self.total_time_ms() <= 50.0


@dataclass
class LayerTask:
    """一层的一次计算任务"""
    layer_id: int  # 1-5
    priority: int  # 0=最高
    deadline_ms: float
    fn: Callable
    input_data: Any = None

    def __lt__(self, other):
        # 优先级高 (数值小) 的先执行
        if self.priority != other.priority:
            return self.priority < other.priority
        return self.deadline_ms < other.deadline_ms


class CognitiveScheduler:
    """
    认知周期调度器

    设计原则:
    1. 保证硬截止 — 超时直接中断当前层
    2. 优先级抢占 — 紧急消息跳过队列直通
    3. 资源自适应 — 根据 GPU 利用率动态调整每层预算
    4. 无全局锁 — 层间通过消息队列通信, 非共享内存
    """

    CYCLE_MS = 50.0
    LAYER_DEADLINES = {
        1: 5.0,
        2: 15.0,
        3: 30.0,
        4: 45.0,
        5: 50.0,
    }
    LAYER_PRIORITIES = {
        1: 0,  # 最高 — 传感器
        2: 1,  # 感知-行动
        3: 2,  # 世界模型
        4: 3,  # 因果推理
        5: 4,  # 元认知 (最低, 可跳)
    }

    def __init__(
        self,
        l1_runner, l2_runner, l3_runner, l4_runner, l5_runner,
        device: str = "cuda",
    ):
        self.runners = {
            1: l1_runner, 2: l2_runner, 3: l3_runner,
            4: l4_runner, 5: l5_runner,
        }

        self.device = device

        # 消息队列
        self.downward_queues: Dict[int, deque] = {
            i: deque(maxlen=100) for i in range(1, 6)
        }
        self.upward_queues: Dict[int, deque] = {
            i: deque(maxlen=100) for i in range(1, 6)
        }

        # 状态
        self.cycle_count = 0
        self.emergency = False
        self.surprise_threshold = 10.0  # L2 surprise > 此值 → 紧急

        # 统计
        self.metrics_history: List[CycleMetrics] = []
        self.avg_layer_times = {i: 0.0 for i in range(1, 6)}

    async def run(self):
        """主循环 — 永远在 50ms 周期内"""
        while True:
            cycle_start = time.perf_counter()

            # 1. 确定模式
            mode = self._determine_mode()

            # 2. 准备任务队列
            tasks = self._build_task_list(mode)

            # 3. 执行 (时间限制内, 按优先级)
            metrics = await self._execute_cycle(tasks, mode, cycle_start)

            # 4. 周期结束处理
            elapsed = (time.perf_counter() - cycle_start) * 1000
            metrics.layer_times = self._last_layer_times

            self.metrics_history.append(metrics)
            self.cycle_count += 1

            # 5. 等待下一个周期 (若有剩余时间)
            if elapsed < self.CYCLE_MS:
                await asyncio.sleep((self.CYCLE_MS - elapsed) / 1000)

            # 6. 日志 (每 100 周期)
            if self.cycle_count % 100 == 0:
                self._log_stats()

    def _determine_mode(self) -> CycleMode:
        """根据 surprise 等级和最新感知决定周期模式"""
        if self.emergency or self._latest_surprise > self.surprise_threshold:
            return CycleMode.URGENT

        # 检查是否有新的感知数据
        has_new_sensing = any(
            not q or q[-1].timestamp > self._last_sensing_time
            for q in self.upward_queues.values()
        )
        if not has_new_sensing:
            return CycleMode.IDLE

        # 每 10 周期触发一次反思
        if self.cycle_count % 10 == 9:
            return CycleMode.REFLECTIVE

        return CycleMode.NORMAL

    def _build_task_list(self, mode: CycleMode) -> List[LayerTask]:
        """
        构建本周期的任务列表

        不同模式选择不同的层:
        - NORMAL:  L1→L2→L3→L4→L5 (全五层)
        - URGENT:  L1→L2  (反射弧)
        - IDLE:    L3  ((内部模拟, 多步预测)
        - REFLECTIVE: L5 (用 L5 独占整个 200ms 长周期)
        """
        tasks = []

        if mode == CycleMode.NORMAL:
            layers = [1, 2, 3, 4, 5]
        elif mode == CycleMode.URGENT:
            layers = [1, 2]
        elif mode == CycleMode.IDLE:
            layers = [3]
        elif mode == CycleMode.REFLECTIVE:
            layers = [5]
        else:
            layers = [1, 2, 3, 4, 5]

        for lid in layers:
            deadline = self.LAYER_DEADLINES[lid]
            priority = self.LAYER_PRIORITIES[lid]

            # 构建层的运行函数
            async def make_runner(layer_id):
                async def run():
                    # 收集来自下方层的消息
                    msgs = self._drain_upward(layer_id)
                    # 运行为
                    result = await self.runners[layer_id].step(
                        msgs,
                        deadline_ms=deadline,
                        budget_ms=self._dynamic_budget(layer_id),
                    )
                    # 输出推到目标队列
                    for msg in result.messages:
                        if msg.direction == "up":
                            self._push_upward(layer_id + 1, msg)
                        elif msg.direction == "down":
                            self._push_downward(layer_id - 1, msg)
                    return result
                return run

            tasks.append(LayerTask(
                layer_id=lid,
                priority=priority,
                deadline_ms=deadline,
                fn=make_runner(lid),
            ))

        # 按优先级排序
        tasks.sort()
        return tasks

    async def _execute_cycle(
        self, tasks: List[LayerTask], mode: CycleMode, cycle_start: float
    ) -> CycleMetrics:
        """
        在时间限制内执行任务列表

        策略: 顺序执行 (各层有依赖), 但每层有独立超时
        """
        metrics = CycleMetrics(
            cycle_id=self.cycle_count,
            start_time=cycle_start,
            mode=mode,
        )
        self._last_layer_times = {}

        cycle_deadline = cycle_start + self.CYCLE_MS / 1000

        for task in tasks:
            now = time.perf_counter()
            remaining_ms = (cycle_deadline - now) * 1000

            if remaining_ms <= 1.0 and task.layer_id > 2:
                # 时间耗尽 — 跳过低优先级层
                logger.debug(f"Cycle {self.cycle_count}: skipping L{task.layer_id} (timeout)")
                metrics.messages_dropped += 1
                continue

            # 执行该层, 带超时
            layer_start = time.perf_counter()
            try:
                await asyncio.wait_for(
                    task.fn(),
                    timeout=min(task.deadline_ms, remaining_ms) / 1000,
                )
                layer_elapsed = (time.perf_counter() - layer_start) * 1000
            except asyncio.TimeoutError:
                layer_elapsed = task.deadline_ms
                logger.warning(
                    f"L{task.layer_id} timed out @ {layer_elapsed:.1f}ms"
                )

            self._last_layer_times[task.layer_id] = layer_elapsed
            self._update_avg_time(task.layer_id, layer_elapsed)
            metrics.messages_sent += 1

        # GPU 利用率
        if torch.cuda.is_available():
            metrics.gpu_util_pct = torch.cuda.utilization()

        metrics.surprise = self._latest_surprise
        return metrics

    def _dynamic_budget(self, layer_id: int) -> float:
        """
        根据历史平均和当前全局负载动态分配时间预算

        公式: budget = base · (1 + α·(ideal_avg - actual_avg)/ideal_avg)
        """
        base = self.LAYER_DEADLINES[layer_id]
        actual_avg = self.avg_layer_times.get(layer_id, base)
        if actual_avg < 1e-3:
            return base
        alpha = 0.3
        adjusted = base * (1 + alpha * (base - actual_avg) / base)
        return max(1.0, min(adjusted, base * 1.5))

    def _drain_upward(self, layer_id: int) -> List:
        """取出下方的所有待处理消息"""
        msgs = []
        for lid in range(1, layer_id):
            q = self.upward_queues[lid]
            while q:
                msgs.append(q.popleft())
        return msgs

    def _push_upward(self, layer_id: int, msg):
        self.upward_queues[layer_id].append(msg)

    def _push_downward(self, layer_id: int, msg):
        self.downward_queues[layer_id].append(msg)

    def trigger_emergency(self):
        """外部触发: 高 surprise → 下周期进入紧急模式"""
        self.emergency = True
        logger.info("EMERGENCY mode activated")

    def clear_emergency(self):
        self.emergency = False
        logger.info("EMERGENCY cleared, resuming NORMAL")

    def _update_avg_time(self, layer_id: int, elapsed: float):
        """指数移动平均"""
        prev = self.avg_layer_times[layer_id]
        self.avg_layer_times[layer_id] = 0.9 * prev + 0.1 * elapsed

    @property
    def _latest_surprise(self) -> float:
        """从 L2 的最新误差获取 surprise (由 PCN trainer 写入)"""
        return getattr(self.runners[2], "latest_surprise", 0.0)

    def _log_stats(self):
        total_ms = sum(self.avg_layer_times.values())
        recent = self.metrics_history[-100:]
        deadlines_met = sum(1 for m in recent if m.deadline_met())
        logger.info(
            f"Cycle {self.cycle_count}: "
            f"avg_total={total_ms:.1f}ms, "
            f"deadlines_met={deadlines_met}/100, "
            f"gpu={recent[-1].gpu_util_pct:.0f}%"
        )
```

```python
# src/layer_runner.py

from dataclasses import dataclass, field
from typing import List


@dataclass
class ProtocolMessage:
    """消息协议 v1.0 消息格式"""
    msg_type: str  # PRED↓ / ERROR↑ / QUERY⇅ / RESULT⇅ / META_OVERRIDE↓ / ...
    source_layer: int
    target_layer: int
    direction: str  # "up" / "down"
    precision: float  # 精度权重
    payload: dict
    timestamp: float = 0.0


@dataclass
class LayerResult:
    messages: List[ProtocolMessage]
    metrics: dict
    surprise: float = 0.0


class LayerRunner:
    """
    层运行器基类

    每层实现 step() — 被调度器在每个周期调用
    """

    def __init__(self, layer_id: int, device: str = "cuda"):
        self.layer_id = layer_id
        self.device = device
        self.latest_surprise = 0.0

    async def step(
        self,
        input_messages: List[ProtocolMessage],
        deadline_ms: float,
        budget_ms: float,
    ) -> LayerResult:
        raise NotImplementedError


# 具体示例: L3 世界模型层
class L3WorldModelRunner(LayerRunner):
    """L3: 神经场世界模型"""

    def __init__(self, world_model, pcn_trainer, device="cuda"):
        super().__init__(layer_id=3, device=device)
        self.model = world_model
        self.trainer = pcn_trainer
        self.internal_steps = 0

    async def step(self, input_messages, deadline_ms, budget_ms):
        messages = []

        # 处理来自 L2 的 ERROR↑ 消息 (新的感知残差)
        for msg in input_messages:
            if msg.msg_type == "ERROR↑" and msg.precision > 0.1:
                # 更新世界模型以最小化预测误差
                self.trainer.step(
                    prediction=msg.payload.get("pred"),
                    observation=msg.payload.get("obs"),
                    precision=msg.precision,
                )

                # 推送全局更新到 L4 (新的概念状态)
                messages.append(ProtocolMessage(
                    msg_type="GLOBAL⇄",
                    source_layer=3,
                    target_layer=4,
                    direction="up",
                    precision=msg.precision,
                    payload={"slots": self.model.get_object_slots()},
                ))

        # 如果是空闲周期: 内部模拟 (多步预测)
        if not input_messages:
            for _ in range(min(10, int(budget_ms / 5))):
                self.model.predict_next_step()
                self.internal_steps += 1

        # L3 的 surprise = 当前预测误差的 norm
        surprise = self.trainer.current_error_norm()
        self.latest_surprise = surprise

        # 向下推送预测 (供 L2 使用)
        predictions = self.model.get_predictions()
        for l2_target in [2]:
            messages.append(ProtocolMessage(
                msg_type="PRED↓",
                source_layer=3,
                target_layer=l2_target,
                direction="down",
                precision=self.trainer.prediction_confidence(),
                payload={"fields": predictions},
            ))

        return LayerResult(messages=messages, metrics={
            "internal_steps": self.internal_steps,
        }, surprise=surprise)
```

```python
class EmergencyReflex:
    """
    L1→L2 短路: 高 surprise → 直接行动, 不经过 L3-L5

    这是公理 A10 (伦理前置) 的工程体现:
    生存关键决策不依赖复杂推理 — 宁可保守不可分析瘫痪
    """

    REFLEX_ACTIONS = {
        "collision_imminent": {"action": "BRAKE", "urgency": 1.0},
        "free_fall": {"action": "STABILIZE", "urgency": 1.0},
        "overheat": {"action": "THROTTLE_DOWN", "urgency": 0.9},
        "obstacle_close": {"action": "EVADE", "urgency": 0.8},
    }

    def __init__(self, scheduler: CognitiveScheduler):
        self.scheduler = scheduler

    def check_and_react(self, l1_data: dict) -> Optional[dict]:
        """
        L1 数据直通 — 不经过任何高层推理

        Returns:
            若触发反射则返回紧急动作; 否则 None (走正常流程)
        """
        # 碰撞预警: 深度/距离传感器
        if l1_data.get("min_distance", float("inf")) < 0.1:
            self.scheduler.trigger_emergency()
            return self.REFLEX_ACTIONS["collision_imminent"]

        # 自由落体: 加速度计突跳
        if l1_data.get("accel_magnitude", 0) > 15.0:
            self.scheduler.trigger_emergency()
            return self.REFLEX_ACTIONS["free_fall"]

        return None  # 无反射 — 正常五层处理
```

```python
class ResourceMonitor:
    """跟踪 GPU/CPU 使用率并调整各层预算"""

    def __init__(self):
        self.gpu_memory_budget = {
            1: 0.05,  # 5% 传感器
            2: 0.15,  # 15% 感知
            3: 0.40,  # 40% 世界模型 (最大)
            4: 0.25,  # 25% 因果推理
            5: 0.15,  # 15% 元认知
        }

    def adjust_for_load(self, current_util: float):
        """如果 GPU 已达 90%, 缩减轻量任务"""
        if current_util > 90:
            # 减 L5 (元认知) 预算, 给 L3 留空间
            self.gpu_memory_budget[5] = max(0.02, self.gpu_memory_budget[5] * 0.7)
            self.gpu_memory_budget[3] += 0.05

    def get_layer_budget_gb(self, layer_id: int, total_mem_gb: float):
        return self.gpu_memory_budget[layer_id] * total_mem_gb
```

### 4.2 优先级继承

低优先级的 L3 任务如果被高优先级的 L2 任务等待，则提升优先级：

```python
def priority_inheritance(tasks, blocked_on):
    """若 L2 被 L3 阻塞, L3 继承 L2 的高优先级"""
    for task in tasks:
        deps = blocked_on.get(task.layer_id, [])
        for dep_id in deps:
            dep_task = next(t for t in tasks if t.layer_id == dep_id)
            dep_task.priority = min(dep_task.priority, task.priority)
```

| 场景 | 期望行为 |
|------|---------|
| L3 计算出 50ms 超限 | 中断 L3 → 继续 L2 行动 (过时的世界模型好过无行动) |
| GPU OOM | L5→L4→L3 降级 (释放内存), 保 L1+L2 |
| 连续 3 周期超时 | 发 META_ALERT↑ → L5 诊断 (是模型复杂度问题还是硬件问题) |
| L1 传感器掉线 | 用 L3 预测器作为虚拟传感器, 发 PRIOR↓ 到 L2 |
| 高负载 + 紧急同时 | 紧急优先; 普通认知放下一周期 |
